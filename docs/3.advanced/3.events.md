---
title: Events
description: React to roles and permissions being attached or detached.
navigation:
  icon: i-tabler-radio
---

Events are disabled by default:

```ts [config/permissions.ts]
defineConfig({
  // ...
  events: { enabled: true },
})
```

| Event                            | Payload                   |
| -------------------------------- | ------------------------- |
| `permission:role_attached`       | `RoleAttachedEvent`       |
| `permission:role_detached`       | `RoleDetachedEvent`       |
| `permission:permission_attached` | `PermissionAttachedEvent` |
| `permission:permission_detached` | `PermissionDetachedEvent` |

Every payload has `model`, the `role` / `permission` (model or name) and the `teamId`.

```ts [start/events.ts]
import emitter from '@adonisjs/core/services/emitter'

emitter.on('permission:role_attached', (event) => {
  console.log(`Role ${event.role} attached to`, event.model.$primaryKeyValue)
})
```
